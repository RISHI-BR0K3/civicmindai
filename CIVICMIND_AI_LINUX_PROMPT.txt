1. FastAPI + SQLite backend with JWT authentication (Citizen, Manager, Admin roles)
2. Scikit-learn AI: NLP text classification (routing issues to departments) and DBSCAN spatial clustering (detecting complaint hotspots)
3. React + Vite + Tailwind CSS frontend with React-Leaflet GIS mapping, HTML5 Geolocation API, OpenStreetMap reverse geocoding to human-readable street addresses, and live webcam + photo upload capability

from sqlalchemy import Boolean, Column, ForeignKey, Integer, String, Float, DateTime
from sqlalchemy.orm import relationship
import datetime
from .database import Base

class Department(Base):
    __tablename__ = "departments"
    id = Column(Integer, primary_key=True, index=True)
    name = Column(String, unique=True, index=True)
    
    users = relationship("User", back_populates="department")
    complaints = relationship("Complaint", back_populates="department")

class User(Base):
    __tablename__ = "users"
    id = Column(Integer, primary_key=True, index=True)
    name = Column(String)
    first_name = Column(String, nullable=True)
    last_name = Column(String, nullable=True)
    contact_number = Column(String, nullable=True)
    occupation = Column(String, nullable=True)
    email = Column(String, unique=True, index=True)
    password_hash = Column(String)
    role = Column(String, default="CITIZEN") # CITIZEN, OFFICER, HEAD, ADMIN
    department_id = Column(Integer, ForeignKey("departments.id"), nullable=True)
    
    department = relationship("Department", back_populates="users")
    complaints = relationship("Complaint", back_populates="citizen")

class Complaint(Base):
    __tablename__ = "complaints"
    id = Column(Integer, primary_key=True, index=True)
    citizen_id = Column(Integer, ForeignKey("users.id"))
    description = Column(String)
    address = Column(String, nullable=True)
    image_url = Column(String, nullable=True)
    latitude = Column(Float)
    longitude = Column(Float)
    department_id = Column(Integer, ForeignKey("departments.id"), nullable=True)
    status = Column(String, default="Pending") # Pending, In Progress, Resolved
    created_at = Column(DateTime, default=datetime.datetime.utcnow)
    
    citizen = relationship("User", back_populates="complaints")
    department = relationship("Department", back_populates="complaints")

class Hotspot(Base):
    __tablename__ = "hotspots"
    id = Column(Integer, primary_key=True, index=True)
    center_latitude = Column(Float)
    center_longitude = Column(Float)
    complaint_count = Column(Integer)
    severity = Column(String)
    created_at = Column(DateTime, default=datetime.datetime.utcnow)

import os
import pickle
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.naive_bayes import MultinomialNB
from sklearn.pipeline import make_pipeline

from ..models import Department

MODEL_PATH = "nlp_model.pkl"

TRAINING_DATA = [
    ("huge pothole on the road", "Roads"),
    ("street light is broken", "Electricity"),
    ("power outage in my area", "Electricity"),
    ("water pipe burst leaking water everywhere", "Water & Sanitation"),
    ("garbage is not collected for days", "Waste Management"),
    ("road is cracked and dangerous", "Roads"),
    ("drainage is overflowing on the street", "Water & Sanitation")
]

def train_and_save_model():
    texts = [item[0] for item in TRAINING_DATA]
    labels = [item[1] for item in TRAINING_DATA]
    
    model = make_pipeline(TfidfVectorizer(), MultinomialNB())
    model.fit(texts, labels)
    
    with open(MODEL_PATH, "wb") as f:
        pickle.dump(model, f)
    
    return model

def load_model():
    if os.path.exists(MODEL_PATH):
        with open(MODEL_PATH, "rb") as f:
            return pickle.load(f)
    else:
        return train_and_save_model()

nlp_model = load_model()

def predict_department(text: str, db):
    predicted_dept_name = nlp_model.predict([text])[0]
    
    department = db.query(Department).filter(Department.name == predicted_dept_name).first()
    
    if not department:
        department = Department(name=predicted_dept_name)
        db.add(department)
        db.commit()
        db.refresh(department)
        
    return department.id

from sklearn.cluster import DBSCAN
import numpy as np
import datetime
from sqlalchemy.orm import Session
from ..models import Complaint, Hotspot

def detect_hotspots(db: Session):
    complaints = db.query(Complaint).filter(Complaint.status != "Resolved").all()
    
    if len(complaints) < 3:
        return
        
    coords = np.array([[c.latitude, c.longitude] for c in complaints])
    
    dbscan = DBSCAN(eps=0.005, min_samples=3)
    clusters = dbscan.fit_predict(coords)
    
    db.query(Hotspot).delete()
    
    unique_labels = set(clusters)
    for label in unique_labels:
        if label == -1:
            continue
            
        cluster_mask = (clusters == label)
        cluster_coords = coords[cluster_mask]
        
        center_lat = np.mean(cluster_coords[:, 0])
        center_lng = np.mean(cluster_coords[:, 1])
        count = len(cluster_coords)
        
        severity = "High" if count >= 5 else "Medium"
        
        new_hotspot = Hotspot(
            center_latitude=center_lat,
            center_longitude=center_lng,
            complaint_count=count,
            severity=severity
        )
        db.add(new_hotspot)
        
    db.commit()

from fastapi import APIRouter, Depends, HTTPException, status
from sqlalchemy.orm import Session
from pydantic import BaseModel
import bcrypt
import jwt
import datetime

from ..database import get_db
from ..models import User

router = APIRouter()

SECRET_KEY = "super_secret_key_for_college_project"
ALGORITHM = "HS256"

class UserCreate(BaseModel):
    first_name: str = ""
    last_name: str = ""
    contact_number: str = ""
    occupation: str = ""
    email: str
    password: str
    name: str = None
    role: str = "CITIZEN"
    department_id: int = None

class UserLogin(BaseModel):
    email: str
    password: str

def get_password_hash(password):
    return bcrypt.hashpw(password.encode('utf-8'), bcrypt.gensalt()).decode('utf-8')

def verify_password(plain_password, hashed_password):
    return bcrypt.checkpw(plain_password.encode('utf-8'), hashed_password.encode('utf-8'))

def create_access_token(data: dict):
    to_encode = data.copy()
    expire = datetime.datetime.utcnow() + datetime.timedelta(hours=24)
    to_encode.update({"exp": expire})
    encoded_jwt = jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)
    return encoded_jwt

@router.post("/register")
def register(user: UserCreate, db: Session = Depends(get_db)):
    db_user = db.query(User).filter(User.email == user.email).first()
    if db_user:
        raise HTTPException(status_code=400, detail="Email already registered")
    
    hashed_password = get_password_hash(user.password)
    full_name = f"{user.first_name} {user.last_name}".strip() or user.name or user.email
    new_user = User(
        name=full_name, 
        first_name=user.first_name,
        last_name=user.last_name,
        contact_number=user.contact_number,
        occupation=user.occupation,
        email=user.email, 
        password_hash=hashed_password, 
        role=user.role, 
        department_id=user.department_id
    )
    db.add(new_user)
    db.commit()
    db.refresh(new_user)

    access_token = create_access_token(data={
        "sub": new_user.email,
        "role": new_user.role,
        "id": new_user.id,
        "dept_id": new_user.department_id
    })

    return {
        "message": "User created successfully",
        "access_token": access_token,
        "token_type": "bearer",
        "role": new_user.role
    }

@router.post("/login")
def login(user: UserLogin, db: Session = Depends(get_db)):
    db_user = db.query(User).filter(User.email == user.email).first()
    if not db_user or not verify_password(user.password, db_user.password_hash):
        raise HTTPException(status_code=401, detail="Invalid credentials")
    
    access_token = create_access_token(data={"sub": db_user.email, "role": db_user.role, "id": db_user.id, "dept_id": db_user.department_id})
    return {"access_token": access_token, "token_type": "bearer", "role": db_user.role}

from fastapi import APIRouter, Depends, HTTPException, Header
from sqlalchemy.orm import Session
from pydantic import BaseModel
import jwt
from ..database import get_db
from ..models import Complaint, User
from .auth import SECRET_KEY, ALGORITHM
from ..ai.nlp_classifier import predict_department

router = APIRouter()

class ComplaintCreate(BaseModel):
    description: str
    address: str = None
    latitude: float
    longitude: float
    image_url: str = None

class ComplaintUpdate(BaseModel):
    status: str

def get_current_user(authorization: str = Header(None), db: Session = Depends(get_db)):
    if not authorization or not authorization.startswith("Bearer "):
        raise HTTPException(status_code=401, detail="Missing or invalid token")
    
    token = authorization.split(" ")[1]
    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        email = payload.get("sub")
        user = db.query(User).filter(User.email == email).first()
        if not user:
            raise HTTPException(status_code=401, detail="User not found")
        return user
    except jwt.PyJWTError:
        raise HTTPException(status_code=401, detail="Invalid token")

@router.post("/")
def create_complaint(complaint: ComplaintCreate, db: Session = Depends(get_db), current_user: User = Depends(get_current_user)):
    predicted_dept_id = predict_department(complaint.description, db)
    
    new_complaint = Complaint(
        citizen_id=current_user.id,
        description=complaint.description,
        address=complaint.address,
        latitude=complaint.latitude,
        longitude=complaint.longitude,
        image_url=complaint.image_url,
        department_id=predicted_dept_id
    )
    db.add(new_complaint)
    db.commit()
    db.refresh(new_complaint)
    
    from ..ai.hotspot_detector import detect_hotspots
    detect_hotspots(db)
    
    return {"message": "Complaint created", "complaint": new_complaint}

@router.get("/")
def get_complaints(db: Session = Depends(get_db), current_user: User = Depends(get_current_user)):
    if current_user.role == "CITIZEN":
        complaints = db.query(Complaint).filter(Complaint.citizen_id == current_user.id).all()
    elif current_user.role in ["OFFICER", "HEAD", "ADMIN"]:
        complaints = db.query(Complaint).all()
    else:
        complaints = []
        
    result = []
    for c in complaints:
        citizen_info = None
        if c.citizen:
            citizen_info = {
                "name": c.citizen.name or f"{c.citizen.first_name or ''} {c.citizen.last_name or ''}".strip() or "Citizen",
                "email": c.citizen.email,
                "contact_number": c.citizen.contact_number,
                "occupation": c.citizen.occupation
            }
        result.append({
            "id": c.id,
            "citizen_id": c.citizen_id,
            "citizen": citizen_info,
            "description": c.description,
            "address": c.address,
            "image_url": c.image_url,
            "latitude": c.latitude,
            "longitude": c.longitude,
            "status": c.status,
            "created_at": c.created_at.isoformat() if c.created_at else None
        })
    return result

import React, { createContext, useContext, useState, useCallback } from 'react';

const ToastContext = createContext();

export function ToastProvider({ children }) {
  const [toasts, setToasts] = useState([]);

  const addToast = useCallback((message, type = 'success') => {
    const id = Date.now();
    setToasts(prev => [...prev, { id, message, type }]);
    setTimeout(() => {
      setToasts(prev => prev.filter(t => t.id !== id));
    }, 3000);
  }, []);

  return (
    <ToastContext.Provider value={addToast}>
      {children}
      <div className="fixed bottom-4 right-4 z-50 flex flex-col space-y-2">
        {toasts.map(t => (
          <div 
            key={t.id} 
            className={`px-4 py-3 rounded shadow-lg text-white text-sm font-bold animate-fade-in ${t.type === 'error' ? 'bg-red-600' : 'bg-green-600'}`}
          >
            {t.message}
          </div>
        ))}
      </div>
    </ToastContext.Provider>
  );
}

export function useToast() {
  return useContext(ToastContext);
}

import { Eye, EyeOff, Loader2 } from 'lucide-react';
import CitizenDashboard from './pages/CitizenDashboard';
import OfficerDashboard from './pages/OfficerDashboard';
import AdminDashboard from './pages/AdminDashboard';
import api from './api';
import { ToastProvider, useToast } from './ToastContext';

function Auth() {
  const [isSignUp, setIsSignUp] = useState(false);
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [firstName, setFirstName] = useState('');
  const [lastName, setLastName] = useState('');
  const [contactNumber, setContactNumber] = useState('');
  const [occupation, setOccupation] = useState('Resident');
  const [showPassword, setShowPassword] = useState(false);
  const [isLoading, setIsLoading] = useState(false);
  const toast = useToast();

  const handleLogin = async (e) => {
    e.preventDefault();
    setIsLoading(true);
    try {
      const response = await api.post('/auth/login', { email, password });
      localStorage.setItem('token', response.data.access_token);
      localStorage.setItem('role', response.data.role);
      
      toast("Login successful! Redirecting...");
      setTimeout(() => {
        if (response.data.role === 'CITIZEN') window.location.href = '/citizen';
        else if (response.data.role === 'OFFICER') window.location.href = '/officer';
        else window.location.href = '/admin';
      }, 500);
    } catch (err) {
      toast("Login failed! Please check credentials.", "error");
      setIsLoading(false);
    }
  };

  const handleSignUp = async (e) => {
    e.preventDefault();
    if (password.length < 6) {
      toast("Password must be at least 6 characters.", "error");
      return;
    }
    setIsLoading(true);
    try {
      const response = await api.post('/auth/register', {
        first_name: firstName,
        last_name: lastName,
        contact_number: contactNumber,
        occupation: occupation,
        email: email,
        password: password
      });

      if (response.data.access_token) {
        localStorage.setItem('token', response.data.access_token);
        localStorage.setItem('role', response.data.role);
        toast(`Welcome to CivicMind AI, ${firstName || 'Citizen'}!`);
        setTimeout(() => {
          window.location.href = '/citizen';
        }, 600);
      } else {
        toast("Account created! Please sign in.");
        setIsSignUp(false);
        setIsLoading(false);
      }
    } catch (err) {
      const msg = err.response?.data?.detail || "Registration failed. Please try again.";
      toast(msg, "error");
      setIsLoading(false);
    }
  };

  return (
    <div className="flex flex-col items-center justify-center my-6">
      <div className="text-center mb-6">
        <h1 className="text-4xl md:text-5xl font-extrabold text-blue-900 mb-2 tracking-tight">CIVICMIND AI</h1>
        <p className="text-lg text-gray-600 italic font-medium">"Predict • Prioritize • Prevent"</p>
      </div>

      <div className={`w-full ${isSignUp ? 'max-w-lg' : 'max-w-sm'} bg-white p-8 border border-gray-200 rounded-2xl shadow-sm transition-all`}>
        <div className="flex border-b border-gray-200 mb-6">
          <button
            type="button"
            onClick={() => setIsSignUp(false)}
            className={`flex-1 py-2 text-center font-bold text-sm border-b-2 transition ${!isSignUp ? 'border-blue-600 text-blue-600' : 'border-transparent text-gray-500 hover:text-gray-700'}`}
          >
            Sign In
          </button>
          <button
            type="button"
            onClick={() => setIsSignUp(true)}
            className={`flex-1 py-2 text-center font-bold text-sm border-b-2 transition ${isSignUp ? 'border-blue-600 text-blue-600' : 'border-transparent text-gray-500 hover:text-gray-700'}`}
          >
            Create Account
          </button>
        </div>

        {!isSignUp ? (
          <form onSubmit={handleLogin} autoComplete="off">
            <div className="mb-4">
              <label className="block text-gray-700 text-sm font-bold mb-2">Email Address</label>
              <input 
                type="email" 
                value={email} 
                onChange={e => setEmail(e.target.value)} 
                required 
                autoComplete="off" 
                placeholder="you@example.com"
                className="w-full border border-gray-300 p-2.5 rounded-lg focus:outline-none focus:border-blue-500 focus:ring-1 focus:ring-blue-500" 
              />
            </div>
            <div className="mb-6 relative">
              <label className="block text-gray-700 text-sm font-bold mb-2">Password</label>
              <div className="relative">
                <input 
                  type={showPassword ? "text" : "password"} 
                  value={password} 
                  onChange={e => setPassword(e.target.value)} 
                  required 
                  autoComplete="new-password" 
                  placeholder="••••••••"
                  className="w-full border border-gray-300 p-2.5 pr-10 rounded-lg focus:outline-none focus:border-blue-500 focus:ring-1 focus:ring-blue-500" 
                />
                <button 
                  type="button" 
                  onClick={() => setShowPassword(!showPassword)} 
                  className="absolute inset-y-0 right-0 pr-3 flex items-center text-gray-500 hover:text-gray-700 focus:outline-none"
                >
                  {showPassword ? <EyeOff size={20} /> : <Eye size={20} />}
                </button>
              </div>
            </div>
            
            <div className="flex flex-col space-y-3">
              <button 
                type="submit" 
                disabled={isLoading}
                className="w-full bg-blue-600 text-white font-bold py-2.5 px-4 rounded-lg hover:bg-blue-700 transition shadow-sm disabled:bg-blue-400 flex justify-center items-center"
              >
                {isLoading ? <Loader2 className="animate-spin mr-2" size={20} /> : null}
                {isLoading ? "Signing in..." : "Sign In"}
              </button>
              <div className="relative flex py-2 items-center">
                <div className="flex-grow border-t border-gray-200"></div>
                <span className="flex-shrink-0 mx-4 text-gray-400 text-xs uppercase">New to CivicMind?</span>
                <div className="flex-grow border-t border-gray-200"></div>
              </div>
              <button 
                type="button" 
                onClick={() => setIsSignUp(true)}
                className="w-full bg-gray-50 border border-gray-300 text-gray-700 font-bold py-2.5 px-4 rounded-lg hover:bg-gray-100 transition shadow-sm"
              >
                Create an Account
              </button>
            </div>
          </form>
        ) : (
          <form onSubmit={handleSignUp} autoComplete="off">
            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4 mb-4">
              <div>
                <label className="block text-gray-700 text-xs font-bold uppercase mb-1">First Name *</label>
                <input 
                  type="text" 
                  value={firstName} 
                  onChange={e => setFirstName(e.target.value)} 
                  required 
                  placeholder="e.g. John"
                  className="w-full border border-gray-300 p-2 rounded-lg text-sm focus:outline-none focus:border-blue-500" 
                />
              </div>
              <div>
                <label className="block text-gray-700 text-xs font-bold uppercase mb-1">Last Name *</label>
                <input 
                  type="text" 
                  value={lastName} 
                  onChange={e => setLastName(e.target.value)} 
                  required 
                  placeholder="e.g. Doe"
                  className="w-full border border-gray-300 p-2 rounded-lg text-sm focus:outline-none focus:border-blue-500" 
                />
              </div>
            </div>

            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4 mb-4">
              <div>
                <label className="block text-gray-700 text-xs font-bold uppercase mb-1">Contact Number *</label>
                <input 
                  type="tel" 
                  value={contactNumber} 
                  onChange={e => setContactNumber(e.target.value)} 
                  required 
                  placeholder="+91 98765 43210"
                  className="w-full border border-gray-300 p-2 rounded-lg text-sm focus:outline-none focus:border-blue-500" 
                />
              </div>
              <div>
                <label className="block text-gray-700 text-xs font-bold uppercase mb-1">Occupation *</label>
                <select 
                  value={occupation} 
                  onChange={e => setOccupation(e.target.value)}
                  className="w-full border border-gray-300 p-2 rounded-lg text-sm bg-white focus:outline-none focus:border-blue-500"
                >
                  <option value="Resident">Local Resident</option>
                  <option value="Student">Student</option>
                  <option value="IT / Software Professional">IT / Software Professional</option>
                  <option value="Business Owner / Merchant">Business Owner / Merchant</option>
                  <option value="Healthcare Worker">Healthcare Worker</option>
                  <option value="Educator / Teacher">Educator / Teacher</option>
                  <option value="Civil Engineer">Civil Engineer</option>
                  <option value="Government Employee">Government Employee</option>
                  <option value="Social Worker">Social Worker</option>
                  <option value="Other">Other</option>
                </select>
              </div>
            </div>

            <div className="mb-4">
              <label className="block text-gray-700 text-xs font-bold uppercase mb-1">Email Address *</label>
              <input 
                type="email" 
                value={email} 
                onChange={e => setEmail(e.target.value)} 
                required 
                placeholder="john.doe@example.com"
                className="w-full border border-gray-300 p-2 rounded-lg text-sm focus:outline-none focus:border-blue-500" 
              />
            </div>

            <div className="mb-6 relative">
              <label className="block text-gray-700 text-xs font-bold uppercase mb-1">Password *</label>
              <div className="relative">
                <input 
                  type={showPassword ? "text" : "password"} 
                  value={password} 
                  onChange={e => setPassword(e.target.value)} 
                  required 
                  autoComplete="new-password" 
                  placeholder="Min 6 characters"
                  className="w-full border border-gray-300 p-2 pr-10 rounded-lg text-sm focus:outline-none focus:border-blue-500" 
                />
                <button 
                  type="button" 
                  onClick={() => setShowPassword(!showPassword)} 
                  className="absolute inset-y-0 right-0 pr-3 flex items-center text-gray-500 hover:text-gray-700 focus:outline-none"
                >
                  {showPassword ? <EyeOff size={18} /> : <Eye size={18} />}
                </button>
              </div>
              <p className="text-gray-400 text-xs mt-1">Make sure it has at least 6 characters.</p>
            </div>

            <div className="flex flex-col space-y-3">
              <button 
                type="submit" 
                disabled={isLoading}
                className="w-full bg-blue-600 text-white font-bold py-2.5 px-4 rounded-lg hover:bg-blue-700 transition shadow-sm disabled:bg-blue-400 flex justify-center items-center"
              >
                {isLoading ? <Loader2 className="animate-spin mr-2" size={20} /> : null}
                {isLoading ? "Creating Account..." : "Complete Registration"}
              </button>
              <div className="text-center text-xs text-gray-500 pt-2">
                Already registered?{' '}
                <button 
                  type="button" 
                  onClick={() => setIsSignUp(false)}
                  className="text-blue-600 hover:underline font-bold"
                >
                  Sign In instead
                </button>
              </div>
            </div>
          </form>
        )}
      </div>
    </div>
  );
}

function App() {
  const isAuthenticated = !!localStorage.getItem('token');

  const logout = () => {
    localStorage.removeItem('token');
    localStorage.removeItem('role');
    window.location.href = '/';
  };

  return (
    <ToastProvider>
      <Router>
        <div className="min-h-screen flex flex-col bg-gray-50">
          <header className="bg-blue-800 text-white shadow-md">
            <div className="max-w-6xl mx-auto px-4 py-3 flex justify-between items-center">
              <Link to="/" className="hover:text-blue-200 transition">
                <h1 className="text-xl font-bold">CivicMind AI</h1>
              </Link>
              <nav className="space-x-6 flex items-center">
                {isAuthenticated && <button onClick={logout} className="hover:text-blue-200 transition">Logout</button>}
              </nav>
            </div>
          </header>
          <main className="flex-grow w-full max-w-6xl mx-auto p-4 mt-6">
            <Routes>
              <Route path="/" element={<Auth />} />
              <Route path="/citizen" element={<CitizenDashboard />} />
              <Route path="/officer" element={<OfficerDashboard />} />
              <Route path="/admin" element={<AdminDashboard />} />
            </Routes>
          </main>
        </div>
      </Router>
    </ToastProvider>
  );
}

export default App;

import React, { useState, useEffect, useRef } from 'react';
import { MapContainer, TileLayer, Marker, useMapEvents, useMap } from 'react-leaflet';
import { Loader2 } from 'lucide-react';
import 'leaflet/dist/leaflet.css';
import api from '../api';
import { useToast } from '../ToastContext';

import L from 'leaflet';
import markerIcon2x from 'leaflet/dist/images/marker-icon-2x.png';
import markerIcon from 'leaflet/dist/images/marker-icon.png';
import markerShadow from 'leaflet/dist/images/marker-shadow.png';

delete L.Icon.Default.prototype._getIconUrl;
L.Icon.Default.mergeOptions({
  iconUrl: markerIcon,
  iconRetinaUrl: markerIcon2x,
  shadowUrl: markerShadow,
});

function LocationPicker({ position, setPosition }) {
  const map = useMapEvents({
    click(e) {
      setPosition(e.latlng);
    },
  });

  useEffect(() => {
    if (position) {
      map.flyTo(position, 15, { animate: true });
    }
  }, [position, map]);

  return position === null ? null : <Marker position={position}></Marker>;
}

export default function CitizenDashboard() {
  const [complaints, setComplaints] = useState([]);
  const [description, setDescription] = useState('');
  const [position, setPosition] = useState(null);
  const [isCameraOpen, setIsCameraOpen] = useState(false);
  const [photoUrl, setPhotoUrl] = useState(null);
  const [isSubmitting, setIsSubmitting] = useState(false);
  const toast = useToast();

  const videoRef = useRef(null);
  const canvasRef = useRef(null);

  const [address, setAddress] = useState('');

  useEffect(() => {
    if (position) {
      const getAddress = async () => {
        try {
          const res = await fetch(`https://nominatim.openstreetmap.org/reverse?format=json&lat=${position.lat}&lon=${position.lng}`);
          const data = await res.json();
          setAddress(data.display_name || "Unknown Location");
        } catch (error) {
          setAddress("Could not fetch address");
        }
      };
      getAddress();
    }
  }, [position]);

  const [expandedId, setExpandedId] = useState(null);

  const toggleExpand = (id) => {
    setExpandedId(expandedId === id ? null : id);
  };

  const defaultCenter = [28.6139, 77.2090]; 

  const fetchComplaints = async () => {
    try {
      const res = await api.get('/complaints/');
      setComplaints(res.data);
    } catch (e) {
      toast("Failed to load history", "error");
    }
  };

  useEffect(() => { fetchComplaints(); }, []);

  const startCamera = async () => {
    setIsCameraOpen(true);
    setPhotoUrl(null);
    try {
      const stream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: 'environment' } });
      if (videoRef.current) videoRef.current.srcObject = stream;
    } catch (err) {
      toast("Camera access denied or no camera found.", "error");
      setIsCameraOpen(false);
    }
  };

  const stopCamera = () => {
    if (videoRef.current && videoRef.current.srcObject) {
      videoRef.current.srcObject.getTracks().forEach(track => track.stop());
    }
    setIsCameraOpen(false);
  };

  const takePhoto = () => {
    const video = videoRef.current;
    const canvas = canvasRef.current;
    if (video && canvas) {
      canvas.width = video.videoWidth;
      canvas.height = video.videoHeight;
      canvas.getContext('2d').drawImage(video, 0, 0, canvas.width, canvas.height);
      setPhotoUrl(canvas.toDataURL('image/png'));
      stopCamera();
    }
  };

  const handleFileUpload = (e) => {
    const file = e.target.files[0];
    if (file) {
      const reader = new FileReader();
      reader.onload = (event) => setPhotoUrl(event.target.result);
      reader.readAsDataURL(file);
    }
  };

  const handleDetectLocation = (e) => {
    e.preventDefault();
    if (navigator.geolocation) {
      toast("Requesting location access...");
      navigator.geolocation.getCurrentPosition(
        (loc) => {
          setPosition({ lat: loc.coords.latitude, lng: loc.coords.longitude });
          toast("Location detected!");
        },
        () => toast("Could not detect location. Please ensure permissions are enabled.", "error")
      );
    } else {
      toast("Geolocation is not supported by your browser.", "error");
    }
  };

  const isFormValid = description.trim().length >= 10 && position !== null;

  const handleSubmit = async (e) => {
    e.preventDefault();
    if (!isFormValid) return;
    
    setIsSubmitting(true);
    try {
      await api.post('/complaints/', {
        description,
        address: address,
        latitude: parseFloat(position.lat),
        longitude: parseFloat(position.lng),
        image_url: photoUrl
      });
      toast('Complaint successfully submitted!');
      setDescription('');
      setPosition(null);
      setPhotoUrl(null);
      setAddress('');
      fetchComplaints();
    } catch (e) {
      toast("Error submitting complaint.", "error");
    } finally {
      setIsSubmitting(false);
    }
  };

  return (
    <div className="max-w-6xl mx-auto grid grid-cols-1 md:grid-cols-2 gap-6">
      <div className="bg-white p-6 border border-gray-200 rounded shadow-sm">
        <h2 className="text-xl font-bold mb-4 text-gray-800 border-b pb-2">Report an Issue</h2>
        <form onSubmit={handleSubmit}>
          <div className="mb-4">
            <label className="block text-gray-700 text-sm font-bold mb-2">Description</label>
            <textarea value={description} onChange={(e) => setDescription(e.target.value)} className="w-full border p-2 rounded focus:outline-none focus:border-blue-500" rows="3" placeholder="E.g. Broken water pipe leaking on Main Street..." required></textarea>
            {description.length > 0 && description.trim().length < 10 && (
              <p className="text-red-500 text-xs mt-1 font-semibold">Please describe the issue in more detail (at least 10 characters).</p>
            )}
          </div>
          
          <div className="mb-6 border p-4 rounded bg-gray-50">
            <label className="block text-gray-700 text-sm font-bold mb-2">Attach Photo Proof (Optional)</label>
            {!isCameraOpen && !photoUrl && (
              <div className="flex space-x-4">
                <button type="button" onClick={startCamera} className="bg-green-600 text-white px-4 py-2 rounded text-sm hover:bg-green-700 transition font-semibold">📷 Take Photo</button>
                <label className="bg-gray-600 text-white px-4 py-2 rounded text-sm hover:bg-gray-700 transition cursor-pointer font-semibold">
                  📁 Upload File
                  <input type="file" accept="image/*" onChange={handleFileUpload} className="hidden" />
                </label>
              </div>
            )}
            
            {isCameraOpen && (
              <div className="mt-2 flex flex-col items-center">
                <video ref={videoRef} autoPlay className="w-full max-h-48 bg-black rounded" />
                <div className="flex space-x-2 mt-2">
                  <button type="button" onClick={takePhoto} className="bg-blue-600 text-white px-4 py-2 rounded font-bold hover:bg-blue-700">Snap!</button>
                  <button type="button" onClick={stopCamera} className="bg-red-600 text-white px-4 py-2 rounded font-bold hover:bg-red-700">Cancel</button>
                </div>
              </div>
            )}
            <canvas ref={canvasRef} className="hidden" />

            {photoUrl && !isCameraOpen && (
              <div className="mt-2 relative inline-block">
                <img src={photoUrl} alt="Complaint" className="max-h-48 rounded border shadow-sm" />
                <button type="button" onClick={() => setPhotoUrl(null)} className="absolute top-1 right-1 bg-red-600 text-white text-xs font-bold px-2 py-1 rounded">Remove</button>
              </div>
            )}
          </div>
          
          <div className="mb-6">
            <div className="flex justify-between items-center mb-2">
              <label className="block text-gray-700 text-sm font-bold">Pinpoint Location on Map *</label>
              <button type="button" onClick={handleDetectLocation} className="bg-blue-100 border border-blue-300 text-blue-800 hover:bg-blue-200 px-3 py-1 rounded text-sm font-bold transition">📍 Detect My Location</button>
            </div>
            <div className={`h-64 w-full rounded border overflow-hidden ${!position ? 'border-red-300' : 'border-green-400 border-2'}`}>
              <MapContainer center={defaultCenter} zoom={5} style={{ height: '100%', width: '100%' }}>
                <TileLayer attribution='&copy; OpenStreetMap' url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png" />
                <LocationPicker position={position} setPosition={setPosition} />
              </MapContainer>
            </div>
            {!position && <p className="text-red-500 text-xs mt-1 font-semibold">You must click the map or detect your location to submit.</p>}
            
            {address && (
              <p className="text-sm text-gray-700 mt-2 bg-gray-100 p-2 rounded border">
                <strong>Address:</strong> {address}
              </p>
            )}
          </div>
          
          <button 
            type="submit" 
            disabled={!isFormValid || isSubmitting}
            className="w-full bg-blue-600 text-white font-bold py-3 px-4 rounded hover:bg-blue-700 transition disabled:bg-gray-400 disabled:cursor-not-allowed flex justify-center items-center"
          >
            {isSubmitting ? <Loader2 className="animate-spin mr-2" size={20} /> : null}
            {isSubmitting ? "Analyzing & Submitting..." : "Submit Complaint"}
          </button>
        </form>
      </div>

      <div className="bg-white p-6 border border-gray-200 rounded shadow-sm">
        <h2 className="text-xl font-bold mb-4 text-gray-800 border-b pb-2">My History</h2>
        <p className="text-gray-500 text-sm mb-4">Click "View Details" to see the full report of your submitted issues.</p>
        <div className="overflow-x-auto">
          <table className="min-w-full text-left text-sm">
            <thead className="uppercase border-b-2 bg-gray-50 text-gray-600">
              <tr>
                <th className="px-4 py-3 font-semibold">ID</th>
                <th className="px-4 py-3 font-semibold">Issue Snippet</th>
                <th className="px-4 py-3 font-semibold">Status</th>
                <th className="px-4 py-3 font-semibold text-right">Action</th>
              </tr>
            </thead>
            <tbody>
              {complaints.map((c) => (
                <React.Fragment key={c.id}>
                  <tr className={`border-b hover:bg-gray-50 transition-colors ${expandedId === c.id ? 'bg-blue-50' : ''}`}>
                    <td className="px-4 py-4 font-bold text-gray-800">#{c.id}</td>
                    <td className="px-4 py-4 text-gray-800 truncate max-w-xs">{c.description}</td>
                    <td className="px-4 py-4">
                      <span className={`px-2 py-1 rounded text-xs font-bold ${
                        c.status === 'Pending' ? 'bg-yellow-100 text-yellow-800' :
                        c.status === 'Resolved' ? 'bg-green-100 text-green-800' : 'bg-blue-100 text-blue-800'
                      }`}>
                        {c.status}
                      </span>
                    </td>
                    <td className="px-4 py-4 text-right">
                      <button 
                        onClick={() => toggleExpand(c.id)} 
                        type="button"
                        className="text-blue-600 hover:text-blue-800 font-semibold text-sm underline"
                      >
                        {expandedId === c.id ? 'Hide Details' : 'View Details'}
                      </button>
                    </td>
                  </tr>

                  {expandedId === c.id && (
                    <tr className="border-b bg-gray-50">
                      <td colSpan="4" className="px-6 py-6">
                        <div className="bg-white border rounded shadow-sm p-6 grid grid-cols-1 md:grid-cols-2 gap-6">
                          <div className="space-y-4">
                            <div>
                              <h4 className="text-xs uppercase font-bold text-gray-400 tracking-wider">Date Submitted</h4>
                              <p className="text-gray-800 font-medium">
                                {new Date(c.created_at + 'Z').toLocaleString('en-IN', {
                                  day: '2-digit', month: '2-digit', year: 'numeric',
                                  hour: '2-digit', minute: '2-digit',
                                  hour12: false
                                })}
                              </p>
                            </div>
                            
                            <div>
                              <h4 className="text-xs uppercase font-bold text-gray-400 tracking-wider">Full Description</h4>
                              <p className="text-gray-800 bg-gray-50 p-3 border rounded mt-1">{c.description}</p>
                            </div>

                        <button onClick={() => deleteComplaint(c.id)} className="text-white bg-red-600 hover:bg-red-700 font-bold text-xs px-3 py-1 rounded shadow-sm">Delete</button>
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
            
            {complaints.length > 10 && (
              <div className="mt-4 flex justify-between items-center bg-gray-50 p-3 rounded border">
                <span className="text-sm text-gray-600 font-semibold">Showing top 10 recent records (Pagination enabled)</span>
                <div className="space-x-2">
                  <button disabled className="px-3 py-1 border rounded bg-gray-200 text-gray-400 cursor-not-allowed text-sm font-bold">Prev</button>
                  <button disabled className="px-3 py-1 border rounded bg-gray-200 text-gray-400 cursor-not-allowed text-sm font-bold">Next</button>
                </div>
              </div>
            )}
          </div>
        )}
      </div>
    </div>
  );
}
